{# The rows of one cluster's proxy table, in one place because they are rendered from two: the page, and the refresh that replaces them while something is on its way. A second copy of this markup would drift from the first, and the drift would show as a row that changes shape when it updates. #} {% macro proxy_rows(cluster, role, unpublished, queued, written, foreign, lamps) %} {% set drafted = unpublished | map(attribute='object_id') | list %} {# Drafts first, then objects written but not deployed, then the rest. The published list is long enough that an alphabetical position would hide the one row that changed, and a row joins that order the moment the nodes are serving it - which is why these rows are replaced in place rather than left until somebody reloads. #} {% for draft in unpublished %} {{ draft.object_id }}—{% if draft.object_id in queued %}Waiting for publish{% else %}{% if draft.deleted %}Draft: removal{% else %}Draft{% endif %}{% endif %}——— {% endfor %} {# One row per object, under the name its author gave it. Not one per frontend: the generator names those after the address and protocol they bind, and an https object that redirects plain HTTP takes two of them. #} {% for object in written if object.name not in drafted %} {% if role == 'admin' %}{{ object.name }}{% else %}{{ object.name }}{% endif %} {% if object.protocol %}{{ object.protocol | upper }}{% else %}—{% endif %} {{ object.hostnames | join(', ') if object.hostnames else 'No hostname rule' }} · {{ object.ports | join(', ') if object.ports else '—' }}{% if not object.readable %}Cannot be read{% elif not object.serving %}Waiting for deploy{% endif %} {# The nodes' own verdict on this object's servers, not a probe. An unlit lamp is an object with no health check: HAProxy reports an unchecked server as up, so a green one here would claim a reachability nothing measured. Grey is a check nobody has reported on yet - an object waiting for a deploy, or a node running an older status script. #} {% set lamp = lamps.get(object.name) if lamps else none %}{% if lamp %}{{ lamp.label }}{% else %}—{% endif %} {% if object.backend_count is not none %}{{ object.backend_count }}{% else %}—{% endif %} Managed {% endfor %} {# And what this platform does not own. Here the frontend is the only thing there is to show, so it is shown as itself. #} {% for proxy in foreign %} {% if role == 'admin' %}{{ proxy.name }}{% else %}{{ proxy.name }}{% endif %}{% if not proxy.enabled %}Disabled{% endif %}{{ proxy.protocol | upper }}{{ proxy.hostnames | join(', ') if proxy.hostnames else 'No hostname rule' }} · {{ proxy.ports | join(', ') if proxy.ports else '—' }}—{{ proxy.backend_count }}Unmanaged {% endfor %} {% endmacro %}
